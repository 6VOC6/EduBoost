-- ==============================
-- CREAR BASE DE DATOS
-- ==============================

CREATE DATABASE EduBoost;
GO

USE EduBoost;
GO


-- ==============================
-- TABLA USUARIOS
-- ==============================

CREATE TABLE Usuarios(
    IdUsuario INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Correo NVARCHAR(100) UNIQUE NOT NULL,
    Password NVARCHAR(255) NOT NULL,
    Rol NVARCHAR(50) NOT NULL,
    FechaRegistro DATETIME DEFAULT GETDATE()
);
GO


-- ==============================
-- TABLA CURSOS
-- ==============================

CREATE TABLE Cursos(
    IdCurso INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuarioAsesor INT NULL, -- El asesor que creó el curso
    Nombre NVARCHAR(150) NOT NULL,
    Descripcion NVARCHAR(500),
    Profesor NVARCHAR(100), 
    FechaCreacion DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Cursos_Asesor
    FOREIGN KEY (IdUsuarioAsesor)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE SET NULL
);
GO


-- ==============================
-- TABLA INSCRIPCIONES
-- ==============================

CREATE TABLE Inscripciones(
    IdInscripcion INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario INT NOT NULL,
    IdCurso INT NOT NULL,
    FechaInscripcion DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Inscripciones_Usuarios
    FOREIGN KEY (IdUsuario)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE CASCADE,

    CONSTRAINT FK_Inscripciones_Cursos
    FOREIGN KEY (IdCurso)
    REFERENCES Cursos(IdCurso)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA MATERIAL CURSO
-- ==============================

CREATE TABLE MaterialCurso(
    IdMaterial INT IDENTITY(1,1) PRIMARY KEY,
    IdCurso INT NOT NULL,
    Titulo NVARCHAR(200),
    TipoMaterial NVARCHAR(50),
    UrlVideo NVARCHAR(500),

    CONSTRAINT FK_Material_Curso
    FOREIGN KEY (IdCurso)
    REFERENCES Cursos(IdCurso)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA PROGRESO
-- ==============================

CREATE TABLE Progreso(
    IdProgreso INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario INT NOT NULL,
    IdCurso INT NOT NULL,
    Porcentaje INT DEFAULT 0,
    UltimaActualizacion DATETIME,

    CONSTRAINT FK_Progreso_Usuario
    FOREIGN KEY (IdUsuario)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE CASCADE,

    CONSTRAINT FK_Progreso_Curso
    FOREIGN KEY (IdCurso)
    REFERENCES Cursos(IdCurso)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA MATERIALES COMPLETADOS
-- ==============================

CREATE TABLE MaterialesCompletados(
    Id INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario INT NOT NULL,
    IdMaterial INT NOT NULL,
    FechaCompletado DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_MatCompletado_Usuario
    FOREIGN KEY (IdUsuario)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE CASCADE,

    CONSTRAINT FK_MatCompletado_Material
    FOREIGN KEY (IdMaterial)
    REFERENCES MaterialCurso(IdMaterial)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA ASESORIAS
-- ==============================

CREATE TABLE Asesorias(
    IdAsesoria INT IDENTITY(1,1) PRIMARY KEY,
    IdEstudiante INT NOT NULL,
    IdAsesor INT NULL, -- Puede ser nulo hasta que se asigne
    IdCurso INT NOT NULL,
    FechaHora DATETIME NOT NULL,
    Tema NVARCHAR(255) NOT NULL,
    Estado NVARCHAR(50) DEFAULT 'Pendiente', -- Pendiente, Aceptada, Rechazada, Completada
    EnlaceReunion NVARCHAR(500),

    CONSTRAINT FK_Asesoria_Estudiante
    FOREIGN KEY (IdEstudiante)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE CASCADE,

    CONSTRAINT FK_Asesoria_Asesor
    FOREIGN KEY (IdAsesor)
    REFERENCES Usuarios(IdUsuario),

    CONSTRAINT FK_Asesoria_Curso
    FOREIGN KEY (IdCurso)
    REFERENCES Cursos(IdCurso)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA EVALUACIONES
-- ==============================

CREATE TABLE Evaluaciones(
    IdEvaluacion INT IDENTITY(1,1) PRIMARY KEY,
    IdCurso INT NOT NULL,
    Titulo NVARCHAR(200) NOT NULL,
    Descripcion NVARCHAR(500),
    FechaCreacion DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Evaluaciones_Curso
    FOREIGN KEY (IdCurso)
    REFERENCES Cursos(IdCurso)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA PREGUNTAS
-- ==============================

CREATE TABLE Preguntas(
    IdPregunta INT IDENTITY(1,1) PRIMARY KEY,
    IdEvaluacion INT NOT NULL,
    Enunciado NVARCHAR(MAX) NOT NULL,
    OpcionA NVARCHAR(255) NOT NULL,
    OpcionB NVARCHAR(255) NOT NULL,
    OpcionC NVARCHAR(255) NOT NULL,
    RespuestaCorrecta CHAR(1) NOT NULL, -- 'A', 'B' o 'C'

    CONSTRAINT FK_Preguntas_Evaluacion
    FOREIGN KEY (IdEvaluacion)
    REFERENCES Evaluaciones(IdEvaluacion)
    ON DELETE CASCADE
);
GO


-- ==============================
-- TABLA RESULTADOS EVALUACION
-- ==============================

CREATE TABLE ResultadosEvaluacion(
    IdResultado INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario INT NOT NULL,
    IdEvaluacion INT NOT NULL,
    Calificacion DECIMAL(5,2) NOT NULL,
    FechaCompletado DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Resultados_Usuario
    FOREIGN KEY (IdUsuario)
    REFERENCES Usuarios(IdUsuario)
    ON DELETE CASCADE,

    CONSTRAINT FK_Resultados_Evaluacion
    FOREIGN KEY (IdEvaluacion)
    REFERENCES Evaluaciones(IdEvaluacion)
    ON DELETE CASCADE
);
GO
